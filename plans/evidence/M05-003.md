### BFS pathfinding
```
path_from(origin, move_value) -> tuple[str, ...] | None:
    if not may_depart(origin): return None
    budget = 1 if in_nebula else move_value
    queue = [(origin, 0, budget, (origin,))]
    best = {}  # system -> best remaining budget
    while queue:
        current, entered, allowance, route = queue.pop(0)
        if is_gravity_rift(current): allowance += 1
        remaining = allowance - entered
        if best.get(current, -1) >= remaining: continue
        best[current] = remaining
        if remaining <= 0: continue
        neighbours = adjacent(current) + modifiers
        for neighbour in sorted(neighbours):
            if not can_enter(neighbour): continue
            if neighbour == active_system: return route + (neighbour,)
            if not can_pass_through(neighbour, origin): continue
            queue.append((neighbour, entered + 1, allowance, arrived))
    return None
```
