### Benchmark structure
```rust
#[bench]
fn bench_activation_enumeration(b: &mut Bencher) {
    let game = seated_game(...);
    b.iter(|| {
        game.legal_options();
    });
}

#[bench]
fn bench_complete_tactical(b: &mut Bencher) {
    let mut game = seated_game(...);
    b.iter(|| {
        while !game.state.finished {
            let choice = game.legal_options().unwrap();
            let option = SeededRandom(0).choose(choice);
            game.step(choice, option);
        }
    });
}
```
