## CLI version command design

```

  content: 37061c5 (oracle commit)
  rng: PCG32 v0.9

```

### Implementation
```rust
// crates/ti4-cli/src/commands/version.rs
use clap::Args;

#[derive(Args)]
pub struct VersionArgs {}

pub fn handle_version(_args: &VersionArgs) {
    println!("ti4-engine v{}", env!("CARGO_PKG_VERSION"));
    println!("  binary: ti4-cli v{}", env!("CARGO_PKG_VERSION"));
    println!("  git: {}", git_version::git_version().unwrap_or("unknown"));
    println!("  schema: 1.0.0");
    println!("  content: {}", content_version());
    println!("  rng: PCG32 v0.9");
    println!("  rustc: {}", rustc_version::version().unwrap_or("unknown"));
}

fn content_version() -> &'static str {
    // From M00-014 oracle hashes
    "37061c5"
}
```

### Snapshot test
```rust
// crates/ti4-cli/tests/version_snapshot.rs
#[test]
fn version_output_matches_snapshot() {
    let output = std::process::Command::new("ti4")
        .arg("--version")
        .output()
        .expect("failed to execute");
    
    let stdout = String::from_utf8(output.stdout).unwrap();
    
    // Verify structure (not exact match due to git/rustc version)
    assert!(stdout.starts_with("ti4-engine v0.1.0"));
    assert!(stdout.contains("  binary:"));
    assert!(stdout.contains("  git:"));
    assert!(stdout.contains("  schema:"));
    assert!(stdout.contains("  content:"));
    assert!(stdout.contains("  rng:"));
    assert!(stdout.contains("  rustc:"));
}
```
