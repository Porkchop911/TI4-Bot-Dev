### Smoke test structure
```rust
#[test]
fn test_generic_game_terminates() {
    for seed in 0..100 {
        let mut game = seated_game(
            assignments = {"p1": "arborec", "p2": "barony", ...},
            dice_seed = seed,
            cards_per_player = cards_per_player(player_count),
        );
        while !game.state.finished {
            let choice = game.legal_options().unwrap();
            let option = SeededRandom(seed).choose(choice);
            game.step(choice, option);
        }
        assert!(game.state.round <= 20);
    }
}
```
