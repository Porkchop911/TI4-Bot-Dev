```python
def view_to_ndjson(state: GameState, viewer: str) -> dict:
    """Convert GameState to per-player NDJSON view (M00-007b schema)."""
    redacted_players = []
    for p in state.players:
        if p.id == viewer:
            redacted_players.append(player_to_ndjson(p))
        else:
            redacted_players.append(redact_player_to_ndjson(p))
    
    return {
        "type": "view",
        "viewer": viewer,
        "players": redacted_players,
        "systems": [system_to_ndjson(ss) for ss in state.board.systems],
    }

def redact_player_to_ndjson(player: Player) -> dict:
    return {
        "id": player.id,
        "name": player.name,
        "color": player.color,
        "vp": player.victory_points,
        "techs": player.technologies,
        "fleet_capacity": player.fleet_capacity,
        "command_tokens": player.command_tokens,
        "trade_goods": player.trade_goods,
        "home_system": player.home_system,
        "controlled_planets": player.controlled_planets,
        "action_cards_count": len(player.action_cards),
        "action_cards": ["?" for _ in player.action_cards],  # HIDDEN
        "secret_objectives_count": len(player.secret_objectives),
        "secret_objectives": ["?" for _ in player.secret_objectives],  # HIDDEN
        "promissory_notes_count": len(player.promissory_notes),
        "promissory_notes": player.promissory_notes,  # NOT redacted
    }
```
