```python
def state_to_ndjson(state: GameState) -> dict:
    """Convert GameState to NDJSON projection (M00-007a schema)."""
    return {
        "type": "state",
        "turn": state.turn,
        "round": state.round,
        "phase": state.phase.value,
        "active_player": state.active_player,
        "speaker": state.speaker,
        "initiative_order": state.initiative_order,
        "players": [player_to_ndjson(p) for p in state.players],
        "systems": [system_to_ndjson(ss) for ss in state.board.systems],
        "strategy_cards": state.strategy_cards,
        "agenda_phase": state.agenda_phase,
        "agenda_votes": state.agenda_votes,
        "agenda_outcome": state.agenda_outcome,
        "game_over": state.game_over,
        "winner": state.winner,
    }

def player_to_ndjson(player: Player) -> dict:
    return {
        "id": player.id,
        "name": player.name,
        "color": player.color,
        "vp": player.victory_points,
        "techs": player.technologies,
        "fleet_capacity": player.fleet_capacity,
        "command_tokens": player.command_tokens,
        "trade_goods": player.trade_goods,
        "home_system": player.home_system,
        "controlled_planets": player.controlled_planets,
        "action_cards_count": len(player.action_cards),
        "secret_objectives_count": len(player.secret_objectives),
        "promissory_notes_count": len(player.promissory_notes),
    }

def system_to_ndjson(ss: SystemState) -> dict:
    return {
        "id": ss.id,  # from galaxy
        "units": [unit_to_ndjson(u) for u in ss.units],
        "command_tokens": sorted(ss.command_tokens),
        "planet_control": ss.planet_control,
        "planet_units": {
            planet: [unit_to_ndjson(u) for u in units]
            for planet, units in ss.planet_units.items()
        },
    }

def unit_to_ndjson(unit: Unit) -> dict:
    return {
        "type_id": unit.type_id,
        "owner": unit.owner,
        "damage": unit.sustained_damage,
    }
```
