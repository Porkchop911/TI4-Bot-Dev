### Tool: `tools/oracle_integrity_check.py`
```python
#!/usr/bin/env python3
"""Verify oracle repository integrity before migration tests."""

import hashlib
import json
import sys
from pathlib import Path

ORACLE_ROOT = Path(__file__).resolve().parent.parent.parent / "ti4-engine"
HASH_FILE = Path(__file__).resolve().parent.parent / "oracle_hashes.json"

CRITICAL_FILES = [
    "engine/state.py", "engine/choice.py", "engine/game.py",
    "engine/timing.py", "engine/sim.py", "engine/learned_policy.py",
    "engine/policy_linear.py", "engine/views.py", "engine/dice.py",
    "engine/content.py", "engine/galaxy.py", "engine/movement.py",
    "engine/combat.py", "engine/bots.py", "engine/strategy.py",
    # ... all 40 engine files
    # ... all 29 content files
]

def compute_sha256(path: Path) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            h.update(chunk)
    return h.hexdigest()

def check_integrity():
    """Check oracle integrity. Returns True if intact."""
    if not ORACLE_ROOT.exists():
        print(f"ERROR: Oracle root not found: {ORACLE_ROOT}", file=sys.stderr)
        return False
    
    # Check git status (oracle must be clean)
    import subprocess
    result = subprocess.run(
        ["git", "status", "--porcelain"],
        cwd=ORACLE_ROOT, capture_output=True, text=True
    )
    if result.stdout.strip():
        print(f"ERROR: Oracle repository has uncommitted changes!", file=sys.stderr)
        return False
    
    # Check hashes
    with open(HASH_FILE) as f:
        expected = json.load(f)
    
    failures = []
    for rel_path, expected_hash in expected.items():
        full_path = ORACLE_ROOT / rel_path
        if not full_path.exists():
            failures.append(f"{rel_path}: MISSING")
            continue
        actual_hash = compute_sha256(full_path)
        if actual_hash != expected_hash:
            failures.append(f"{rel_path}: HASH MISMATCH (expected={expected_hash[:16]}..., actual={actual_hash[:16]}...)")
    
    if failures:
        print("ERROR: Oracle integrity check failed:", file=sys.stderr)
        for f in failures:
            print(f"  - {f}", file=sys.stderr)
        return False
    
    print("Oracle integrity: PASS")
    return True

if __name__ == "__main__":
    sys.exit(0 if check_integrity() else 1)
```
