### Fixture copy script (tools/copy_fixtures.py)
```python
#!/usr/bin/env python3
"""Copy fixtures from oracle to CI-safe directory."""

import shutil
import sys
from pathlib import Path

ORACLE_ROOT = Path(__file__).resolve().parent.parent.parent / "ti4-engine"
FIXTURES_DIR = Path(__file__).resolve().parent.parent / "fixtures" / "copied"

# Fixtures to copy (from M00-005 artifact inventory)
FIXTURES = [
    "engine/content/manifest.json",
    "engine/content/factions.json",
    "engine/content/systems.json",
    "engine/content/planets.json",
    "engine/content/units.json",
    "engine/content/leaders.json",
    "engine/content/technologies.json",
    "data/map_pools/save52.json.gz",
    "data/map_pools/save54.json.gz",
    "tools/benchmark_engine.py",
    "tools/bot_benchmark.py",
    # ... all 103 selected fixtures from M00-008
]

def copy_fixtures():
    """Copy fixtures from oracle to fixtures/copied/."""
    FIXTURES_DIR.mkdir(parents=True, exist_ok=True)
    
    for rel_path in FIXTURES:
        src = ORACLE_ROOT / rel_path
        dst = FIXTURES_DIR / rel_path
        dst.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(src, dst)
    
    # Generate fixture manifest with hashes
    manifest = {}
    for rel_path in FIXTURES:
        src = ORACLE_ROOT / rel_path
        manifest[str(rel_path)] = sha256(src)
    
    (FIXTURES_DIR / "manifest.json").write_text(json.dumps(manifest, indent=2))

if __name__ == "__main__":
    copy_fixtures()
```

fn oracle_fixture(path: &str) -> PathBuf {
    let dir = env::var("TI4_FIXTURES_DIR")
        .ok()
        .map(PathBuf::from)
        .unwrap_or_else(|_| {
            // Developer mode: copy from oracle
            let oracle = Path::new(env!("CARGO_MANIFEST_DIR"))
                .parent()
                .unwrap()
                .parent()
                .unwrap()
                .join("ti4-engine");
            let fixtures_dir = Path::new(env!("CARGO_MANIFEST_DIR"))
                .join("fixtures")
                .join("copied");
            fixtures_dir
        });
    dir.join(path)
}
