def outcome_to_ndjson(state: GameState, reason: str) -> dict:
    """Convert GameState at game end to NDJSON outcome (M00-007e schema)."""
    return {
        "type": "outcome",
        "game_over": state.game_over,
        "winner": state.winner,
        "victory_points": {p.id: p.victory_points for p in state.players},
        "final_phase": state.phase.value,
        "final_turn": state.turn,
        "final_round": state.round,
        "reason": reason,
