@dataclass(frozen=True)

    active_player: str

## Canonical NDJSON state projection
```json
{
  "type": "state",
  "turn": int,
  "round": int,
  "phase": "strategy" | "action" | "status" | "agenda",
  "active_player": str,
  "speaker": str,
  "initiative_order": [str],
  "players": [
    {
      "id": str,
      "name": str,
      "color": str,
      "vp": int,
      "techs": [str],
      "fleet_capacity": int,
      "command_tokens": int,
      "trade_goods": int,
      "home_system": str,
      "controlled_planets": [str],
      "action_cards_count": int,
      "secret_objectives_count": int,
      "promissory_notes_count": int
    }
  ],
  "systems": [
    {
      "id": str,
      "units": [{"type_id": str, "owner": str, "damage": int}],
      "command_tokens": [str],
      "planet_control": {planet_id: player_id},
      "planet_units": {planet_id: [{type_id, owner, damage}]}
    }
  ],
  "strategy_cards": [str],
  "agenda_phase": int,
  "agenda_votes": {player: vote},
  "agenda_outcome": str | null,
  "game_over": bool,
  "winner": str | null
}
```
