### Benchmark harness (example)
```rust
// crates/ti4-engine/benches/game_benchmark.rs
use criterion::{criterion_group, criterion_main, BenchmarkId, Criterion};
use ti4_engine::game::Game;

fn single_core_game(c: &mut Criterion) {
    let mut group = c.benchmark_group("single_core_game");

    for player_count in [2, 3, 4, 5, 6] {
        group.bench_with_input(
            BenchmarkId::new("player_count", player_count),
            &player_count,
            |b, &players| {
                b.iter(|| {
                    let mut game = Game::new(players, 24_800_001);
                    game.run(4);
                    game
                });
            },
        );
    }

    group.finish();
}

criterion_group!(benches, single_core_game);
criterion_main!(benches);
```
